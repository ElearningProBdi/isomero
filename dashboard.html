function can(action, role, licenceCount, courseOwnerId, currentUserId) { if (!PERMISSIONS[role][action]) return false; // 10 licences rule if ((action === 'canCreateCourse' || action === 'canEditCourse') && (role === 'teacher' || role === 'leader')) { if (licenceCount < MIN_LICENCES_PER_YEAR) return false; } // OWNER ONLY rule - Teacher & Leader only their courses if (action === 'canEditCourse' && (role === 'teacher' || role === 'leader')) { return courseOwnerId === currentUserId; // Can only edit OWN } // Admin can edit/delete all if (role === 'admin') return true; return true; }
